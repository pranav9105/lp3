import random

def partition(arr, low, high, count):
    pivot = arr[high]
    i = low-1
    for j in range(low, high):
        count[0]+=1
        if arr[j] <= pivot:
            i+=1
            arr[i], arr[j] = arr[j], arr[i]
    arr[i+1], arr[high] = arr[high], arr[i+1]
    return i+1

def random_partition(arr, low, high, count):
    pivot = random.randint(low, high)
    arr[pivot], arr[high] = arr[high], arr[pivot]
    return partition(arr, low, high, count)

def deterministic(arr, low, high, count):
    if low<high: 
        pi = partition(arr, low, high, count)
        deterministic(arr, low, pi-1, count)
        deterministic(arr, pi+1, high, count)

def randomized(arr, low, high, count):
    if low<high:
        pi = random_partition(arr, low, high, count)
        randomized(arr, low, pi-1, count)
        randomized(arr, pi+1, high, count)

# 1. What is Quick Sort?
# Quick Sort is a sorting algorithm that arranges elements in ascending or descending order.

# It uses the Divide and Conquer technique.

# Basic logic:
# 1. Select one element as a pivot.
# 2. Place smaller elements on the left side of the pivot.
# 3. Place greater elements on the right side.
# 4. Repeat the same process for the left and right parts.
# 5. Finally, the array becomes sorted.
# There are two types in your program:
# - Deterministic Quick Sort
# - Randomized Quick Sort

# Next, Quick Sort is applied recursively to the left and right subarrays.

# The process continues until the entire array is sorted.
# B. Randomized Quick Sort
# Here, the pivot is selected randomly.
# pivot = random.randint(low, high)



# For example, suppose the randomly selected pivot is 7.

# The program moves that pivot to the last position and performs partitioning.

# The process repeats until the array is sorted.

# Main difference:
# Deterministic Quick Sort	Randomized Quick Sort
# Pivot is fixed (last element in your code).	Pivot is selected randomly.
# Uses partition().	Uses random_partition().
# Performance can be poor for certain input arrangements.	Random pivot reduces the chance of consistently poor partitions.
