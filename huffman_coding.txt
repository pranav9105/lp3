import heapq
from collections import Counter

class Node:
    def __init__(self, char, freq):
        self.char = char
        self.freq = freq
        self.left = None
        self.right = None

def huffman_tree(frequency):
    heap = []
    count = 0
    for char, freq in frequency:
        node = Node(char, freq)
        heapq.heappush(heap, (freq, count, node))
        count += 1

    while len(heap) > 1:
        freq1, _, left = heapq.heappop(heap)
        freq2, _, right = heapq.heappop(heap)
        newNode = Node(None, freq1+freq2)
        newNode.left = left
        newNode.right = right
        heapq.heappush(heap, (newNode.freq, count, newNode))
        count += 1

    return heap[0][2]

def generate_code(root, code, huffman_codes):
    if root is None:
        return
    elif root.char is not None:
        huffman_codes[root.char] = code if code else "0"
        return
    generate_code(root.left, code + "0", huffman_codes)
    generate_code(root.right, code + "1", huffman_codes)

# . What is Huffman Coding?
# Huffman Coding is a lossless data compression technique.

# It reduces the number of bits required to store data by assigning:
# - Short binary codes (0 and 1) to frequently occurring characters.
# - Longer binary codes to less frequently occurring characters.
# For example:

# Suppose we have these characters:
# Character	Frequency
# A	5
# B	9
# C	12
# D	13
# E	16
# F	45

# Here, F occurs 45 times, so it will get a shorter code.
# 2. Logic of Huffman Coding with example
# Step 1: Arrange characters according to frequency (smallest first).

# Our frequencies are:
# A = 5
# B = 9
# C = 12
# D = 13
# E = 16
# F = 45


# Step 2: Take the two smallest frequencies.

# Take A = 5 and B = 9.

# Add them:
# 5 + 9 = 14


# Combine them into a new node with frequency 14.

# Step 3: Repeat the process.

# Now the frequencies are:
# C = 12
# D = 13
# New Node = 14
# E = 16
# F = 45


# Take the two smallest:
# 12 + 13 = 25


# Again, combine them.

# Continue until only one node remains.

# Step 4: Assign binary codes.
# - Left side = 0
# - Right side = 1
# When we reach the original character, its path from the root gives its Huffman code.

# For example, the final codes for this input can be:
# Character	Frequency	Huffman Code
# A	5	1100
# B	9	1101
# C	12	100
# D	13	101
# E	16	111
# F	45	0

# Note: Equivalent Huffman codes may differ depending on how equal-frequency nodes are ordered.
