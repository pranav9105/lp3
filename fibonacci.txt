def iterative(n):
    series = []
    a, b = 0, 1
    series.append(a)
    series.append(b)
    for i in range(2,n):
        c = a+b
        series.append(c)
        a = b
        b = c
    return series

def recursive(n):
    if n == 0:
        return 0
    elif n == 1:
        return 1
    else:
        return recursive(n-1) + recursive(n-2)
