
# BANK CUSTOMER CHURN PREDICTION USING NEURAL NETWORK

# 1. Import Libraries
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

from sklearn.model_selection import train_test_split
from sklearn.preprocessing import LabelEncoder, StandardScaler
from sklearn.metrics import confusion_matrix, classification_report
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score
from sklearn.metrics import roc_curve, roc_auc_score

from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense, Dropout


# 2. Load Dataset
df = pd.read_csv("churn.csv")

print(df.head())
print(df.info())


# 3. Drop Unnecessary Columns
df = df.drop(['RowNumber', 'CustomerId', 'Surname'], axis=1)


# 4. Encode Categorical Variables
le_gender = LabelEncoder()
df['Gender'] = le_gender.fit_transform(df['Gender'])

df = pd.get_dummies(df, columns=['Geography'], drop_first=True)


# 5. Separate Features and Target
X = df.drop('Exited', axis=1)
y = df['Exited']


# 6. Feature Scaling
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)


# 7. Split Dataset into Training and Testing
X_train, X_test, y_train, y_test = train_test_split(
    X_scaled, y, test_size=0.2, random_state=42
)


# 8. Check Target Distribution
sns.countplot(x='Exited', data=df)
plt.title("Customer Churn Distribution")
plt.show()


# 9. Build Neural Network Model
model = Sequential([
    Dense(16, activation='relu', input_dim=X_train.shape[1]),
    Dropout(0.3),
    Dense(8, activation='relu'),
    Dense(1, activation='sigmoid')
])


# 10. Compile Model
model.compile(
    optimizer='adam',
    loss='binary_crossentropy',
    metrics=['accuracy']
)

model.summary()


# 11. Train Model
history = model.fit(
    X_train, y_train,
    epochs=50,
    batch_size=32,
    validation_split=0.2,
    verbose=1
)


# 12. Plot Training and Validation Accuracy and Loss
plt.figure(figsize=(12,5))

plt.subplot(1,2,1)
plt.plot(history.history['accuracy'], label='Train Accuracy')
plt.plot(history.history['val_accuracy'], label='Validation Accuracy')
plt.legend()
plt.title('Model Accuracy')

plt.subplot(1,2,2)
plt.plot(history.history['loss'], label='Train Loss')
plt.plot(history.history['val_loss'], label='Validation Loss')
plt.legend()
plt.title('Model Loss')

plt.show()


# 13. Make Predictions
y_pred_prob = model.predict(X_test)
y_pred = (y_pred_prob > 0.5).astype(int)


# 14. Evaluate Model Performance
acc = accuracy_score(y_test, y_pred)
prec = precision_score(y_test, y_pred)
rec = recall_score(y_test, y_pred)
f1 = f1_score(y_test, y_pred)

print("Accuracy:", acc)
print("Precision:", prec)
print("Recall:", rec)
print("F1-Score:", f1)

print("\nClassification Report:\n", classification_report(y_test, y_pred))


# 15. Plot Confusion Matrix
cm = confusion_matrix(y_test, y_pred)

sns.heatmap(cm, annot=True, fmt='d', cmap='Blues')
plt.xlabel('Predicted')
plt.ylabel('Actual')
plt.title('Confusion Matrix')
plt.show()


# 16. Calculate ROC Curve and AUC
fpr, tpr, _ = roc_curve(y_test, y_pred_prob)
roc_auc = roc_auc_score(y_test, y_pred_prob)


# 17. Plot ROC Curve
plt.plot(fpr, tpr, label='AUC = %.3f' % roc_auc)
plt.plot([0,1], [0,1], linestyle='--')
plt.xlabel('False Positive Rate')
plt.ylabel('True Positive Rate')
plt.title('ROC Curve')
plt.legend()
plt.show()



1. What is the aim of this practical?
The aim is to predict whether a bank customer will leave the bank or continue using its services in the next 6 months.

This is called Customer Churn Prediction.
- Exited = 1 → Customer left the bank.
- Exited = 0 → Customer stayed with the bank.
Example:

Suppose a bank has customer information such as age, credit score, balance, gender and salary. We train a Neural Network using this information to predict whether a customer might leave.

Neural Network	Model that learns patterns from data

Training	Teaching the model using data
Testing	Checking model performance

ReLU activation

Sigmoid activation

Stay or Leave


Important terms:
- Sequential: Creates a model layer by layer.
- Dense: A fully connected neural network layer.
- 16 and 8: Number of neurons.
- ReLU: Activation function used in hidden layers.
- Dropout(0.3): Randomly turns off 30% of neurons during training to reduce overfitting.
- Sigmoid: Converts output into a probability between 0 and 1.
