def placeQueen(board, row, n):
    if row == n:
        return True
    for col in range(n):
        if is_safe(board, row, col, n):
            board[row][col] = 1
            if placeQueen(board, row+1, n):
                return True
            board[row][col] = 0
    return False

def is_safe(board, row, col, n):

    # upper column
    for i in range(row):
        if board[i][col] == 1:
            return False

    # upper-left diagonal
    i, j = row-1, col-1
    while i>=0 and j>=0:
        if board[i][j] == 1:
            return False
        i-=1
        j-=1

    # upper-right diagonal
    i, j = row-1, col+1
    while i>=0 and j<n:
        if board[i][j] == 1:
            return False
        i-=1
        j+=1
    return True

# 6. Important viva questions
# Question	Answer
# What is N-Queens problem?	Placing N queens on an N × N board without attacking each other.
# Which algorithm is used?	Backtracking.
# What is Backtracking?	Trying a solution and going back if it fails.
# What does 0 represent?	Empty position.
# What does 1 represent?	Queen placed.
# What is recursion?	A function calling itself.
# What is the base condition?	row == n.
# Why do we use is_safe()?	To check whether a position is safe.
# Why do we check diagonals?	Queens can attack diagonally.
# Why do we reset board[row][col] = 0?	To undo an incorrect placement.
# What is the time complexity?	Commonly expressed as O(N!), as each row tries available columns with backtracking.
# What is the space complexity?	O(N²) for the board, plus O(N) recursion depth.
