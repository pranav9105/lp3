
#include <iostream>
#include <vector>
using namespace std;

int n;

// Check whether queen can be placed safely
bool isSafe(vector<vector<int>>& board, int row, int col) {

    // Check column
    for (int i = 0; i < row; i++)
        if (board[i][col])
            return false;

    // Check left diagonal
    for (int i = row - 1, j = col - 1;
         i >= 0 && j >= 0; i--, j--)
        if (board[i][j])
            return false;

    // Check right diagonal
    for (int i = row - 1, j = col + 1;
         i >= 0 && j < n; i--, j++)
        if (board[i][j])
            return false;

    return true;
}

// Solve N-Queens using backtracking
bool solve(vector<vector<int>>& board, int row) {

    // All queens are placed
    if (row == n)
        return true;

    // Try every column
    for (int col = 0; col < n; col++) {

        if (isSafe(board, row, col)) {

            // Place queen
            board[row][col] = 1;

            // Move to next row
            if (solve(board, row + 1))
                return true;

            // Backtrack
            board[row][col] = 0;
        }
    }

    return false;
}

int main() {

    cout << "Enter N: ";
    cin >> n;

    vector<vector<int>> board(n, vector<int>(n, 0));

    int firstCol;

    cout << "Enter column for first Queen (1-" << n << "): ";
    cin >> firstCol;

    firstCol--;

    // Place first queen
    board[0][firstCol] = 1;

    // Place remaining queens
    if (solve(board, 1)) {

        cout << "\nN-Queens Matrix:\n";

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++)
                cout << board[i][j] << " ";

            cout << endl;
        }
    }
    else {
        cout << "\nNo solution exists." << endl;
    }

    return 0;
}


P4 – N-Queens (Backtracking): Time: \(O(N \ N!)\), Space: \(O(N^2)\).
