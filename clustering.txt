
# Experiment: K-Means Clustering on Sales Dataset

# 1. Import Libraries
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans


# 2. Load Dataset
df = pd.read_csv("/content/sales_data_sample.csv", encoding='unicode_escape')
df.head()


# 3. Display Basic Information
print(df.info())
print(df.describe())
print(df.columns)


# 4. Select Numerical Columns
df_clean = df.select_dtypes(include=[np.number]).dropna()


# 5. Select Important Features
features = df_clean[['QUANTITYORDERED', 'PRICEEACH', 'SALES', 'ORDERNUMBER']]


# 6. Standardize Features
scaler = StandardScaler()
X_scaled = scaler.fit_transform(features)

print("Data shape after scaling:", X_scaled.shape)


# 7. Find Optimal Number of Clusters Using Elbow Method
inertia = []
K = range(1, 11)

for k in K:
    model = KMeans(n_clusters=k, random_state=42)
    model.fit(X_scaled)
    inertia.append(model.inertia_)


# 8. Plot Elbow Curve
plt.figure(figsize=(8, 5))
plt.plot(K, inertia, marker='o', color='blue')
plt.title("Elbow Method - Optimal K")
plt.xlabel("Number of Clusters (K)")
plt.ylabel("Inertia (Within-Cluster Sum of Squares)")
plt.grid(True)
plt.show()


# 9. Apply K-Means Clustering (K=4)
kmeans = KMeans(n_clusters=4, random_state=42)
df_clean['Cluster'] = kmeans.fit_predict(X_scaled)

# Add cluster labels to original dataset
df['Cluster'] = kmeans.labels_


# 10. Display Cluster Centers
print("Cluster Centers:\n", kmeans.cluster_centers_)


# 11. Visualize Clusters
plt.figure(figsize=(10, 6))
sns.scatterplot(
    x='PRICEEACH',
    y='SALES',
    hue='Cluster',
    data=df,
    palette='viridis'
)

plt.title('K-Means Clustering Visualization')
plt.xlabel('Price Each')
plt.ylabel('Sales')
plt.legend(title='Cluster')
plt.show()


# 12. Display Cluster Summary
cluster_summary = df.groupby('Cluster')[
    ['SALES', 'PRICEEACH', 'QUANTITYORDERED']
].mean()

print("Cluster Summary:\n", cluster_summary)
