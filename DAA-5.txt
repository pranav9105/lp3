
#include <iostream>
#include <cstdlib>
#include <ctime>
using namespace std;

int comparisons = 0;

// Partition for deterministic quick sort
int partition(int arr[], int low, int high) {
    int pivot = arr[high];
    int i = low - 1;

    for (int j = low; j < high; j++) {
        comparisons++;

        if (arr[j] < pivot) {
            i++;
            swap(arr[i], arr[j]);
        }
    }

    swap(arr[i + 1], arr[high]);
    return i + 1;
}

// Deterministic Quick Sort
void quickSort(int arr[], int low, int high) {
    if (low < high) {
        int p = partition(arr, low, high);

        quickSort(arr, low, p - 1);
        quickSort(arr, p + 1, high);
    }
}

// Randomized Partition
int randomPartition(int arr[], int low, int high) {
    int randomIndex = low + rand() % (high - low + 1);

    swap(arr[randomIndex], arr[high]);

    return partition(arr, low, high);
}

// Randomized Quick Sort
void randomizedQuickSort(int arr[], int low, int high) {
    if (low < high) {
        int p = randomPartition(arr, low, high);

        randomizedQuickSort(arr, low, p - 1);
        randomizedQuickSort(arr, p + 1, high);
    }
}

// Display array
void display(int arr[], int n) {
    for (int i = 0; i < n; i++)
        cout << arr[i] << " ";

    cout << endl;
}

int main() {
    int n;

    cout << "Enter number of elements: ";
    cin >> n;

    int arr1[n], arr2[n];

    cout << "Enter array elements: ";
    for (int i = 0; i < n; i++) {
        cin >> arr1[i];
        arr2[i] = arr1[i];
    }

    // Deterministic Quick Sort
    comparisons = 0;

    quickSort(arr1, 0, n - 1);

    cout << "\nDeterministic Quick Sort:\n";
    display(arr1, n);
    cout << "Comparisons: " << comparisons << endl;

    // Randomized Quick Sort
    comparisons = 0;
    srand(time(0));

    randomizedQuickSort(arr2, 0, n - 1);

    cout << "\nRandomized Quick Sort:\n";
    display(arr2, n);
    cout << "Comparisons: " << comparisons << endl;

    return 0;
}
